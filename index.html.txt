    <script>
        const MY_SECRET_KEY = "youssef2026";

        function checkCode() {
            let entered = document.getElementById("passCode").value;
            if (entered === MY_SECRET_KEY) {
                document.getElementById("loginSection").style.display = "none";
                document.getElementById("hubSection").classList.remove("hidden");
                loadServers();
            } else {
                document.getElementById("errorMsg").innerText = "❌ الكود خطأ! حاول مجدداً.";
            }
        }

        function loadServers() {
            fetch('/api/web-servers')
            .then(res => res.json())
            .then(data => {
                let list = document.getElementById("serversList");
                list.innerHTML = "";
                if (!data || data.length === 0) {
                    list.innerHTML = "<p>لا توجد سيرفرات مسجلة حالياً...</p>";
                    return;
                }
                data.forEach(srv => {
                    let card = document.createElement("div");
                    card.className = "server-card " + (srv.tier === "Omega" ? "omega" : "");
                    card.innerHTML = `
                        <b>Tier:</b> ${srv.tier} <br>
                        <b>Loot/Weapon:</b> ${srv.weapon} (${srv.count}) <br>
                        <b>Ping:</b> ${srv.ping} | <b>Players:</b> ${srv.players} <br>
                        <a href="roblox://experiences/start?placeId=0&gameInstanceId=${srv.jobId}" style="color: #00ffaa; text-decoration: none; font-weight: bold;">[ انضمام مباشر للسيرفر 🚀 ]</a>
                    `;
                    list.appendChild(card);
                });
            }).catch(err => {
                document.getElementById("serversList").innerHTML = "<p>خطأ في الاتصال...</p>";
            });
        }
    </script>
